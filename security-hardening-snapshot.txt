import cors from '@fastify/cors';
import helmet from '@fastify/helmet';
import type { FastifyInstance, FastifyRequest } from 'fastify';
import fastifyPlugin from 'fastify-plugin';

import { env } from '@/shared/config/env.js';
import { HTTP_SECURITY } from './http-security.js';
import { CorsOriginNotAllowedError } from './cors-origin-not-allowed.error.js';

function isAllowedOrigin(origin: string | undefined): boolean {
  if (origin === undefined) {
    return true;
  }

  return HTTP_SECURITY.allowedOrigins.has(origin);
}

async function httpSecurityPluginImplementation(fastify: FastifyInstance): Promise<void> {
  await fastify.register(cors, {
    credentials: HTTP_SECURITY.allowCredentials,
    methods: ['GET', 'POST', 'PUT', 'PATCH', 'DELETE', 'OPTIONS'],
    allowedHeaders: ['authorization', 'content-type', 'x-request-id'],
    exposedHeaders: [
      'x-request-id',
      'x-ratelimit-limit',
      'x-ratelimit-remaining',
      'x-ratelimit-reset',
      'retry-after',
    ],
    maxAge: 60,
    origin(origin, callback): void {
      if (isAllowedOrigin(origin)) {
        callback(null, true);
        return;
      }

      callback(new CorsOriginNotAllowedError(), false);
    },
  });

  await fastify.register(helmet, {
    global: true,
    contentSecurityPolicy: {
      directives: {
        defaultSrc: ["'none'"],
        baseUri: ["'none'"],
        frameAncestors: ["'none'"],
        formAction: ["'none'"],
      },
    },
    crossOriginEmbedderPolicy: false,
    referrerPolicy: {
      policy: 'no-referrer',
    },
    frameguard: {
      action: 'deny',
    },
    hsts:
      env.NODE_ENV === 'production'
        ? {
            maxAge: HTTP_SECURITY.hstsMaxAgeSeconds,
            includeSubDomains: true,
            preload: false,
          }
        : false,
  });

  fastify.addHook('onSend', async (request: FastifyRequest, reply, payload): Promise<unknown> => {
    if (request.url.startsWith('/api/v1/auth')) {
      reply.header('Cache-Control', 'no-store, max-age=0');

      reply.header('Pragma', 'no-cache');

      reply.header('Expires', '0');
    }

    return payload;
  });
}

export const httpSecurityPlugin = fastifyPlugin(httpSecurityPluginImplementation, {
  name: 'http-security-plugin',
});

=== LOGIN CONTROLLER ===
import type { FastifyReply, FastifyRequest } from 'fastify';

import { InvalidCredentialsError } from '@/application/errors/invalid-credentials.error.js';
import { OptimisticConcurrencyError } from '@/application/errors/optimistic-concurrency.error.js';
import type {
  LoginUserCommand,
  LoginUserResult,
} from '@/application/use-cases/authentication/login-user.use-case.js';
import {
  createErrorResponse,
  createSuccessResponse,
  type ApiErrorDetail,
} from '@/shared/http/api-response.js';
import { loginRequestSchema, type LoginRequestBody } from './login.schema.js';
import { createRequestAuditContext } from '@/interfaces/http/auditing/request-audit-context.js';

export interface LoginUserExecutor {
  execute(command: LoginUserCommand): Promise<LoginUserResult>;
}

export interface LoginControllerDependencies {
  loginUserUseCase: LoginUserExecutor;
}

interface LoginRouteRequest {
  Body: LoginRequestBody;
}

export function createLoginController(
  dependencies: LoginControllerDependencies,
): (request: FastifyRequest<LoginRouteRequest>, reply: FastifyReply) => Promise<void> {
  return async function loginController(
    request: FastifyRequest<LoginRouteRequest>,
    reply: FastifyReply,
  ): Promise<void> {
    const validationResult = loginRequestSchema.safeParse(request.body);

    if (!validationResult.success) {
      const details: ApiErrorDetail[] = validationResult.error.issues.map(
        (issue): ApiErrorDetail => {
          const field = issue.path.map(String).join('.');

          if (field.length === 0) {
            return {
              code: issue.code,
              message: issue.message,
            };
          }

          return {
            code: issue.code,
            field,
            message: issue.message,
          };
        },
      );

      await reply
        .status(400)
        .send(createErrorResponse('INVALID_REQUEST', 'Request validation failed', details));

      return;
    }

    try {
      const result = await dependencies.loginUserUseCase.execute({
        applicationId: validationResult.data.applicationId,

        email: validationResult.data.email,

        password: validationResult.data.password,

        ...createRequestAuditContext(request),
      });

      await reply.status(200).send(createSuccessResponse(result));
    } catch (error: unknown) {
      if (error instanceof InvalidCredentialsError) {
        await reply
          .status(401)
          .send(createErrorResponse('INVALID_CREDENTIALS', 'Email or password is incorrect'));

        return;
      }

      if (error instanceof OptimisticConcurrencyError) {
        await reply
          .status(409)
          .send(
            createErrorResponse(
              'CONCURRENT_UPDATE',
              'The user account was modified by another operation',
            ),
          );

        return;
      }

      throw error;
    }
  };
}

import type { AuditRequestContext } from '@/application/auditing/audit-request-context.js';
import { InvalidCredentialsError } from '@/application/errors/invalid-credentials.error.js';
import { OptimisticConcurrencyError } from '@/application/errors/optimistic-concurrency.error.js';
import type { AuthenticationTokenPair } from '@/application/models/authentication-token-pair.js';
import type { AccessTokenService } from '@/application/ports/authentication/access-token-service.js';
import type { PasswordHasher } from '@/application/ports/authentication/password-hasher.js';
import type { RefreshTokenService } from '@/application/ports/authentication/refresh-token-service.js';
import type { LoginSessionRepository } from '@/application/ports/repositories/login-session.repository.js';
import type { UserRepository } from '@/application/ports/repositories/user.repository.js';
import type { Clock } from '@/application/ports/system/clock.js';
import type { IdGenerator } from '@/application/ports/system/id-generator.js';
import { EmailAddress } from '@/domain/authentication/value-objects/email-address.js';
import { DomainValidationError } from '@/domain/errors/domain-validation.error.js';
import type { User } from '@/domain/users/entities/user.entity.js';

export interface LoginUserCommand extends AuditRequestContext {
  applicationId: string;
  email: string;
  password: string;
}

export interface LoginUserResult {
  user: {
    userId: string;
    email: string;
    status: 'ACTIVE';
  };

  sessionId: string;
  tokens: AuthenticationTokenPair;
}

export interface LoginUserUseCaseDependencies {
  userRepository: UserRepository;
  loginSessionRepository: LoginSessionRepository;
  passwordHasher: PasswordHasher;
  accessTokenService: AccessTokenService;
  refreshTokenService: RefreshTokenService;
  idGenerator: IdGenerator;
  clock: Clock;
  dummyPasswordHash: string;
  maximumFailedAttempts: number;
  lockDurationMinutes: number;
  accessTokenTtlMinutes: number;
  refreshTokenTtlDays: number;
}

export class LoginUserUseCase {
  private readonly userRepository: UserRepository;

  private readonly loginSessionRepository: LoginSessionRepository;

  private readonly passwordHasher: PasswordHasher;

  private readonly accessTokenService: AccessTokenService;

  private readonly refreshTokenService: RefreshTokenService;

  private readonly idGenerator: IdGenerator;

  private readonly clock: Clock;

  private readonly dummyPasswordHash: string;

  private readonly maximumFailedAttempts: number;

  private readonly lockDurationMilliseconds: number;

  private readonly accessTokenExpiresInSeconds: number;

  private readonly refreshTokenTtlMilliseconds: number;

  public constructor(dependencies: LoginUserUseCaseDependencies) {
    if (
      !Number.isInteger(dependencies.maximumFailedAttempts) ||
      dependencies.maximumFailedAttempts <= 0
    ) {
      throw new RangeError('Maximum failed login attempts must be a positive integer');
    }

    if (
      !Number.isInteger(dependencies.lockDurationMinutes) ||
      dependencies.lockDurationMinutes <= 0
    ) {
      throw new RangeError('Login lock duration must be a positive integer');
    }

    if (
      !Number.isInteger(dependencies.accessTokenTtlMinutes) ||
      dependencies.accessTokenTtlMinutes <= 0
    ) {
      throw new RangeError('Access token TTL must be a positive integer');
    }

    if (
      !Number.isInteger(dependencies.refreshTokenTtlDays) ||
      dependencies.refreshTokenTtlDays <= 0
    ) {
      throw new RangeError('Refresh token TTL must be a positive integer');
    }

    if (dependencies.dummyPasswordHash.trim().length === 0) {
      throw new RangeError('Dummy password hash must not be empty');
    }

    this.userRepository = dependencies.userRepository;

    this.loginSessionRepository = dependencies.loginSessionRepository;

    this.passwordHasher = dependencies.passwordHasher;

    this.accessTokenService = dependencies.accessTokenService;

    this.refreshTokenService = dependencies.refreshTokenService;

    this.idGenerator = dependencies.idGenerator;

    this.clock = dependencies.clock;

    this.dummyPasswordHash = dependencies.dummyPasswordHash;

    this.maximumFailedAttempts = dependencies.maximumFailedAttempts;

    this.lockDurationMilliseconds = dependencies.lockDurationMinutes * 60 * 1_000;

    this.accessTokenExpiresInSeconds = dependencies.accessTokenTtlMinutes * 60;

    this.refreshTokenTtlMilliseconds = dependencies.refreshTokenTtlDays * 24 * 60 * 60 * 1_000;
  }

  public async execute(command: LoginUserCommand): Promise<LoginUserResult> {
    const email = await this.parseEmailOrReject(command.email, command.password);

    const user = await this.userRepository.findByNormalizedEmail(email.normalizedValue);

    const passwordHash = user?.getPasswordHash() ?? this.dummyPasswordHash;

    const passwordMatches = await this.passwordHasher.verify(command.password, passwordHash);

    if (user === null) {
      throw new InvalidCredentialsError();
    }

    const now = this.clock.now();

    if (user.isLocked(now)) {
      throw new InvalidCredentialsError();
    }

    if (!passwordMatches) {
      await this.recordFailedLogin({
        user,
        now,
        command,
      });

      throw new InvalidCredentialsError();
    }

    if (!user.canAuthenticate(now)) {
      throw new InvalidCredentialsError();
    }

    const expectedUserVersion = user.getVersion();

    user.recordSuccessfulLogin(now);

    const sessionId = this.idGenerator.generate();

    const refreshTokenId = this.idGenerator.generate();

    const refreshTokenFamilyId = this.idGenerator.generate();

    const refreshToken = this.refreshTokenService.generate();

    const refreshTokenExpiresAt = new Date(now.getTime() + this.refreshTokenTtlMilliseconds);

    const accessToken = await this.accessTokenService.create({
      claims: {
        userId: user.getId(),
        sessionId,
        applicationId: command.applicationId,
        email: user.getEmail().value,
      },

      issuedAt: now,
    });

    await this.loginSessionRepository.create({
      user,
      expectedUserVersion,

      sessionId,
      applicationId: command.applicationId,

      refreshTokenId,
      refreshTokenFamilyId,
      refreshTokenJti: refreshToken.jti,
      refreshTokenHash: refreshToken.tokenHash,

      createdAt: now,
      sessionExpiresAt: refreshTokenExpiresAt,
      refreshTokenExpiresAt,

      ...(command.requestId === undefined
        ? {}
        : {
            requestId: command.requestId,
          }),

      ...(command.ipAddress === undefined
        ? {}
        : {
            ipAddress: command.ipAddress,
          }),

      ...(command.userAgent === undefined
        ? {}
        : {
            userAgent: command.userAgent,
          }),
    });

    return {
      user: {
        userId: user.getId(),
        email: user.getEmail().value,
        status: 'ACTIVE',
      },

      sessionId,

      tokens: {
        accessToken,

        refreshToken: refreshToken.rawToken,

        tokenType: 'Bearer',

        accessTokenExpiresInSeconds: this.accessTokenExpiresInSeconds,

        refreshTokenExpiresAt,
      },
    };
  }

  private async parseEmailOrReject(rawEmail: string, password: string): Promise<EmailAddress> {
    try {
      return EmailAddress.create(rawEmail);
    } catch (error: unknown) {
      if (!(error instanceof DomainValidationError)) {
        throw error;
      }

      await this.passwordHasher.verify(password, this.dummyPasswordHash);

      throw new InvalidCredentialsError();
    }
  }

  private async recordFailedLogin(input: {
    user: User;
    now: Date;
    command: LoginUserCommand;
  }): Promise<void> {
    if (input.user.getStatus() !== 'ACTIVE' || !input.user.isEmailVerified()) {
      return;
    }

    const expectedVersion = input.user.getVersion();

    const wasLockedBeforeAttempt = input.user.isLocked(input.now);

    input.user.recordFailedLogin({
      maximumAttempts: this.maximumFailedAttempts,

      lockDurationMilliseconds: this.lockDurationMilliseconds,

      now: input.now,
    });

    const accountLocked = !wasLockedBeforeAttempt && input.user.isLocked(input.now);

    try {
      await this.userRepository.recordFailedLogin({
        user: input.user,
        expectedVersion,

        applicationId: input.command.applicationId,

        failedAt: input.now,
        accountLocked,

        ...(input.command.requestId === undefined
          ? {}
          : {
              requestId: input.command.requestId,
            }),

        ...(input.command.ipAddress === undefined
          ? {}
          : {
              ipAddress: input.command.ipAddress,
            }),

        ...(input.command.userAgent === undefined
          ? {}
          : {
              userAgent: input.command.userAgent,
            }),
      });
    } catch (error: unknown) {
      if (!(error instanceof OptimisticConcurrencyError)) {
        throw error;
      }
    }
  }
}

src/application/use-cases/authentication/refresh-session.use-case.ts
import { InvalidRefreshTokenError } from '@/application/errors/invalid-refresh-token.error.js';
import type { AuthenticationTokenPair } from '@/application/models/authentication-token-pair.js';
import type { AccessTokenService } from '@/application/ports/authentication/access-token-service.js';
import type { RefreshTokenService } from '@/application/ports/authentication/refresh-token-service.js';
import type { RefreshSessionRepository } from '@/application/ports/repositories/refresh-session.repository.js';
import type { Clock } from '@/application/ports/system/clock.js';
import type { IdGenerator } from '@/application/ports/system/id-generator.js';

export interface RefreshSessionCommand {
  refreshToken: string;
}

export interface RefreshSessionResult {
  sessionId: string;
  tokens: AuthenticationTokenPair;
}

export interface RefreshSessionUseCaseDependencies {
  repository: RefreshSessionRepository;
  accessTokenService: AccessTokenService;
  refreshTokenService: RefreshTokenService;
  idGenerator: IdGenerator;
  clock: Clock;
  accessTokenTtlMinutes: number;
  refreshTokenTtlDays: number;
}

export class RefreshSessionUseCase {
  private readonly repository: RefreshSessionRepository;
  private readonly accessTokenService: AccessTokenService;
  private readonly refreshTokenService: RefreshTokenService;
  private readonly idGenerator: IdGenerator;
  private readonly clock: Clock;
  private readonly accessTokenExpiresInSeconds: number;
  private readonly refreshTokenTtlMilliseconds: number;

  public constructor(dependencies: RefreshSessionUseCaseDependencies) {
    if (
      !Number.isInteger(dependencies.accessTokenTtlMinutes) ||
      dependencies.accessTokenTtlMinutes <= 0
    ) {
      throw new RangeError('Access token TTL must be a positive integer');
    }

    if (
      !Number.isInteger(dependencies.refreshTokenTtlDays) ||
      dependencies.refreshTokenTtlDays <= 0
    ) {
      throw new RangeError('Refresh token TTL must be a positive integer');
    }

    this.repository = dependencies.repository;
    this.accessTokenService = dependencies.accessTokenService;
    this.refreshTokenService = dependencies.refreshTokenService;
    this.idGenerator = dependencies.idGenerator;
    this.clock = dependencies.clock;
    this.accessTokenExpiresInSeconds = dependencies.accessTokenTtlMinutes * 60;
    this.refreshTokenTtlMilliseconds = dependencies.refreshTokenTtlDays * 24 * 60 * 60 * 1_000;
  }

  public async execute(command: RefreshSessionCommand): Promise<RefreshSessionResult> {
    const rawCurrentToken = command.refreshToken.trim();

    if (rawCurrentToken.length === 0) {
      throw new InvalidRefreshTokenError();
    }

    const rotatedAt = this.clock.now();

    const newRefreshToken = this.refreshTokenService.generate();

    const requestedTokenExpiresAt = new Date(
      rotatedAt.getTime() + this.refreshTokenTtlMilliseconds,
    );

    const rotationResult = await this.repository.rotate({
      currentTokenHash: this.refreshTokenService.hash(rawCurrentToken),
      newTokenId: this.idGenerator.generate(),
      newTokenJti: newRefreshToken.jti,
      newTokenHash: newRefreshToken.tokenHash,
      rotatedAt,
      requestedTokenExpiresAt,
    });

    if (rotationResult.status !== 'ROTATED') {
      throw new InvalidRefreshTokenError();
    }

    const accessToken = await this.accessTokenService.create({
      claims: {
        userId: rotationResult.userId,
        sessionId: rotationResult.sessionId,
        applicationId: rotationResult.applicationId,
        email: rotationResult.userEmail,
      },
      issuedAt: rotatedAt,
    });

    return {
      sessionId: rotationResult.sessionId,
      tokens: {
        accessToken,
        refreshToken: newRefreshToken.rawToken,
        tokenType: 'Bearer',
        accessTokenExpiresInSeconds: this.accessTokenExpiresInSeconds,
        refreshTokenExpiresAt: rotationResult.refreshTokenExpiresAt,
      },
    };
  }
}

=== REFRESH REPOSITORIES ===
src/infrastructure/database/repositories/prisma-refresh-session.repository.ts
import type {
  RefreshSessionRepository,
  RotateSessionRefreshTokenInput,
  RotateSessionRefreshTokenResult,
} from '@/application/ports/repositories/refresh-session.repository.js';
import type { ApplicationPrismaClient } from '../prisma-client.js';

const REUSE_REVOKE_REASON = 'Refresh token reuse detected';

export class PrismaRefreshSessionRepository implements RefreshSessionRepository {
  private readonly prisma: ApplicationPrismaClient;

  public constructor(prisma: ApplicationPrismaClient) {
    this.prisma = prisma;
  }

  public async rotate(
    input: RotateSessionRefreshTokenInput,
  ): Promise<RotateSessionRefreshTokenResult> {
    return this.prisma.$transaction(
      async (transaction) => {
        const currentToken = await transaction.refreshToken.findUnique({
          where: {
            tokenHash: input.currentTokenHash,
          },

          include: {
            session: {
              include: {
                user: true,
              },
            },
          },
        });

        if (currentToken === null) {
          return {
            status: 'INVALID',
          };
        }

        const session = currentToken.session;
        const user = session.user;

        const tokenWasPreviouslyUsed =
          currentToken.usedAt !== null ||
          currentToken.revokedAt !== null ||
          currentToken.replacedByTokenId !== null;

        if (tokenWasPreviouslyUsed) {
          await transaction.refreshToken.updateMany({
            where: {
              familyId: currentToken.familyId,
              revokedAt: null,
            },

            data: {
              revokedAt: input.rotatedAt,
              revokeReason: REUSE_REVOKE_REASON,
            },
          });

          await transaction.session.updateMany({
            where: {
              id: session.id,
              status: 'ACTIVE',
            },

            data: {
              status: 'REVOKED',
              revokedAt: input.rotatedAt,
              revokeReason: REUSE_REVOKE_REASON,
              updatedAt: input.rotatedAt,
            },
          });

          return {
            status: 'REUSE_DETECTED',
          };
        }

        const tokenHasExpired = currentToken.expiresAt.getTime() <= input.rotatedAt.getTime();

        const sessionHasExpired = session.expiresAt.getTime() <= input.rotatedAt.getTime();

        const sessionIsUnavailable = session.status !== 'ACTIVE' || session.revokedAt !== null;

        const userCannotAuthenticate =
          user.deletedAt !== null || user.status !== 'ACTIVE' || !user.emailVerified;

        if (
          tokenHasExpired ||
          sessionHasExpired ||
          sessionIsUnavailable ||
          userCannotAuthenticate
        ) {
          if (sessionHasExpired && session.status === 'ACTIVE') {
            await transaction.session.updateMany({
              where: {
                id: session.id,
                status: 'ACTIVE',
              },

              data: {
                status: 'EXPIRED',
                updatedAt: input.rotatedAt,
              },
            });
          }

          return {
            status: 'INVALID',
          };
        }

        const refreshTokenExpiresAt = new Date(
          Math.min(input.requestedTokenExpiresAt.getTime(), session.expiresAt.getTime()),
        );

        await transaction.refreshToken.create({
          data: {
            id: input.newTokenId,
            sessionId: session.id,
            familyId: currentToken.familyId,
            tokenHash: input.newTokenHash,
            jti: input.newTokenJti,
            expiresAt: refreshTokenExpiresAt,
            createdAt: input.rotatedAt,
          },
        });

        const currentTokenUpdate = await transaction.refreshToken.updateMany({
          where: {
            id: currentToken.id,
            usedAt: null,
            revokedAt: null,
            replacedByTokenId: null,
          },

          data: {
            usedAt: input.rotatedAt,
            replacedByTokenId: input.newTokenId,
          },
        });

        if (currentTokenUpdate.count === 0) {
          await transaction.refreshToken.deleteMany({
            where: {
              id: input.newTokenId,
              usedAt: null,
              replacedByTokenId: null,
            },
          });

          await transaction.refreshToken.updateMany({
            where: {
              familyId: currentToken.familyId,
              revokedAt: null,
            },

            data: {
              revokedAt: input.rotatedAt,
              revokeReason: REUSE_REVOKE_REASON,
            },
          });

          await transaction.session.updateMany({
            where: {
              id: session.id,
              status: 'ACTIVE',
            },

            data: {
              status: 'REVOKED',
              revokedAt: input.rotatedAt,
              revokeReason: REUSE_REVOKE_REASON,
              updatedAt: input.rotatedAt,
            },
          });

          return {
            status: 'REUSE_DETECTED',
          };
        }

        await transaction.session.update({
          where: {
            id: session.id,
          },

          data: {
            lastActivityAt: input.rotatedAt,
            updatedAt: input.rotatedAt,
          },
        });

        return {
          status: 'ROTATED',

          userId: user.id,
          userEmail: user.email,
          applicationId: session.applicationId,
          sessionId: session.id,

          refreshTokenExpiresAt,
        };
      },
      {
        maxWait: 2_000,
        timeout: 5_000,
      },
    );
  }
}
